% Maximum bit rate per channel, from datasheet/data/max-bitrate.csv (values from the specs page;
% build.sh checks each one is still there).
\pgfplotstableread[col sep=comma]{datasheet/data/max-bitrate.csv}\maxrates
\begin{tikzpicture}
  \begin{axis}[
      xbar, scale only axis, width=0.6\linewidth, height=4.2cm, bar width=0.5cm,
      xmin=0, xmax=9, xtick={0,1,...,9}, xlabel={Maximum bit rate (Mbit/s)},
      ytick=data, yticklabels from table={\maxrates}{channel}, y tick label style={font=\small},
      enlarge y limits=0.18, axis x line*=bottom, axis y line*=left,
      xmajorgrids, grid style={rulegray!60},
      nodes near coords, nodes near coords style={font=\small\bfseries},
      point meta=x,
    ]
    \addplot[fill=brandyellow, draw=black] table[x=mbps, y expr=\coordindex] {\maxrates};
  \end{axis}
\end{tikzpicture}
